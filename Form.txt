Private Sub ColorIndicator_Click()
        On Error Resume Next
        Set ColorCell1 = Application.InputBox("Select a cell with the desired fill color.", Type:=8)
        If ColorCell1 Is Nothing Then Exit Sub
        selectedColor = ColorCell1.Interior.Color
        ColorIndicator.BackColor = selectedColor
        Color_Cell.Value = True
        On Error GoTo 0
End Sub

    End If
End Sub

Private Sub SRng_TextBox_AfterUpdate() ' SearchRange1.Address
        Dim tempRange As Range
        On Error Resume Next
        Set tempRange = Range(SRng_TextBox.Value)
        On Error GoTo 0

        If Not tempRange Is Nothing Then
            Set SearchRange1 = tempRange         ' Valid range, so set it to SelectedRange
        Else
            If SRng_TextBox.Value = "No range selected" Then
                Exit Sub
            ElseIf Not SearchRange1 Is Nothing Then
                SRng_TextBox.Value = SearchRange1.Address
                MsgBox "The entered text is not a valid range.", vbExclamation
            Else
                SRng_TextBox.Value = "No range selected"
                MsgBox "The entered text is not a valid range.", vbExclamation
            End If
    End If

End Sub

Private Sub UserForm_Initialize()

    Me.Width = 375 ' Øèðèíà â ïóíêòàõ
    Me.Height = 235 ' Âûñîòà â ïóíêòàõ

    If PackSearch = True Then
        Pack_search.Value = True
    Else
        Pack_search.Value = False
        PackSearch = False
    End If
    
    If colorCell = True Then
        Color_Cell.Value = True
    Else
        Color_Cell.Value = False
        colorCell = False
    End If
   
' selectedColor       Initialize the Value with the previously selected range, if any
    If TypeName(selectedColor) = "Long" And Color_Cell.Value = True Then
        ColorIndicator.BackColor = selectedColor
    Else
        ColorIndicator.BackColor = &H8000000F
    End If
    
    If TypeName(ColorCellChange) = "Boolean" Then
        Color_Cell.Value = ColorCellChange
    Else
        Color_Cell.Value = False
    End If

' SearchRange1       Initialize the TextBox with the previously selected range, if any
    If TypeName(SearchRange1) = "Range" Then
        SRng_TextBox.Value = SearchRange1.Address
    Else
        SRng_TextBox.Value = "No range selected"
    End If
' OutputRange1       Initialize the TextBox with the previously selected range, if any
    If TypeName(OutputRange1) = "Range" Then
        ORng_TextBox.Value = OutputRange1.Address
    Else
        ORng_TextBox.Value = "No range selected"
    End If
 ' Initialize the TextBox with the previously selected range, if any
'DataFileName1 = ""
    If DataFileName1 <> "" Then
        FileN_TextBox.Value = DataFileName1
   Else
        FileN_TextBox.Value = "No File name selected"
    End If
' Initialize the TextBox with the previously selected range, if any
'folderPath1 = ""
    If FolderPath1 <> "" Then
        FileP_TextBox.Value = FolderPath1
    Else
        FileP_TextBox.Value = "No File path selected"
    End If

    With OutputP_list
        .AddItem "Output all values (0)"
        .AddItem "Output PERMANENT values (1)"
        .AddItem "Output Fix-Term values (2)"
        .ListIndex = 0
        OutputFormat1 = 0
    End With


End Sub
